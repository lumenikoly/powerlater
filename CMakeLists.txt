cmake_minimum_required(VERSION 3.24)
if(CMAKE_HOST_APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS version")
endif()
project(PowerTimer VERSION 0.0.1 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

option(POWER_TIMER_BUILD_APP "Build the graphical application" ON)
option(POWER_TIMER_BUILD_TESTS "Build safe, non-destructive tests" ON)
option(POWER_TIMER_ENABLE_CLANG_TIDY "Run clang-tidy while compiling" OFF)
set(POWER_TIMER_FLTK_SOURCE_DIR "" CACHE PATH "Existing FLTK 1.4.5 source tree (offline build)")

if(POWER_TIMER_ENABLE_CLANG_TIDY)
  find_program(CLANG_TIDY_EXECUTABLE NAMES clang-tidy REQUIRED)
  set(CMAKE_CXX_CLANG_TIDY "${CLANG_TIDY_EXECUTABLE}")
endif()

file(GLOB_RECURSE POWER_TIMER_FORMAT_SOURCES CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/src/*.hpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/src/*.mm"
  "${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/tests/*.hpp")
find_program(CLANG_FORMAT_EXECUTABLE NAMES clang-format)
if(CLANG_FORMAT_EXECUTABLE)
  add_custom_target(format
    COMMAND "${CLANG_FORMAT_EXECUTABLE}" -i ${POWER_TIMER_FORMAT_SOURCES}
    COMMENT "Formatting C++ sources")
  add_custom_target(check-format
    COMMAND "${CLANG_FORMAT_EXECUTABLE}" --dry-run --Werror ${POWER_TIMER_FORMAT_SOURCES}
    COMMENT "Checking C++ source formatting")
endif()

function(power_timer_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()

add_library(timer_core STATIC src/timer.cpp)
target_include_directories(timer_core PUBLIC src)
power_timer_warnings(timer_core)

if(WIN32)
  add_library(power_backend STATIC src/power_windows.cpp)
  target_compile_definitions(power_backend PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(power_backend PRIVATE powrprof advapi32 user32)
elseif(APPLE)
  add_library(power_backend STATIC src/power_macos.cpp)
  target_link_libraries(power_backend PRIVATE "-framework Carbon" "-framework ApplicationServices")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_library(process_runner STATIC src/process_linux.cpp)
  target_include_directories(process_runner PUBLIC src)
  target_compile_definitions(process_runner PRIVATE _GNU_SOURCE)
  power_timer_warnings(process_runner)
  add_library(power_backend STATIC src/power_linux.cpp)
  target_link_libraries(power_backend PRIVATE process_runner)
else()
  message(FATAL_ERROR "Supported systems: Windows, macOS, Linux.")
endif()
target_link_libraries(power_backend PUBLIC timer_core)
power_timer_warnings(power_backend)

find_package(Threads REQUIRED)
if(POWER_TIMER_BUILD_TESTS)
  enable_testing()
  add_executable(timer_tests tests/timer_tests.cpp)
  target_link_libraries(timer_tests PRIVATE timer_core)
  power_timer_warnings(timer_tests)
  add_test(NAME timer_logic COMMAND timer_tests)
  add_executable(localization_tests tests/localization_tests.cpp)
  target_link_libraries(localization_tests PRIVATE timer_core)
  power_timer_warnings(localization_tests)
  add_test(NAME localization COMMAND localization_tests)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(process_fixture tests/process_fixture.cpp)
    target_link_libraries(process_fixture PRIVATE Threads::Threads)
    power_timer_warnings(process_fixture)
    add_executable(process_tests tests/process_tests.cpp)
    target_link_libraries(process_tests PRIVATE process_runner)
    power_timer_warnings(process_tests)
    add_test(NAME process_execution COMMAND process_tests $<TARGET_FILE:process_fixture>)
    set_tests_properties(process_execution PROPERTIES TIMEOUT 20)
  endif()
endif()

if(POWER_TIMER_BUILD_APP)
  # Build only the toolkit features used by this two-action utility.
  set(FLTK_BUILD_TEST OFF CACHE BOOL "" FORCE)
  set(FLTK_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(FLTK_BUILD_FLUID OFF CACHE BOOL "" FORCE)
  set(FLTK_BUILD_FLTK_OPTIONS OFF CACHE BOOL "" FORCE)
  set(FLTK_BUILD_GL OFF CACHE BOOL "" FORCE)
  set(FLTK_BUILD_FORMS OFF CACHE BOOL "" FORCE)
  set(FLTK_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(FLTK_MSVC_RUNTIME_DLL OFF CACHE BOOL "" FORCE)
  set(FLTK_OPTION_PRINT_SUPPORT OFF CACHE BOOL "" FORCE)
  set(FLTK_OPTION_SVG OFF CACHE BOOL "" FORCE)
  # X11/XWayland by default: a smaller Linux dependency set, no OpenGL.
  # Native Wayland can be opted into with -DFLTK_BACKEND_WAYLAND=ON.
  set(FLTK_BACKEND_WAYLAND OFF CACHE BOOL "Build native Wayland support")
  if(POWER_TIMER_FLTK_SOURCE_DIR)
    set(fltk_SOURCE_DIR "${POWER_TIMER_FLTK_SOURCE_DIR}")
    add_subdirectory("${POWER_TIMER_FLTK_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/fltk" EXCLUDE_FROM_ALL)
  else()
    include(FetchContent)
    # Upstream FLTK 1.4.5 source release, pinned by the published SHA-256.
    FetchContent_Declare(fltk
      URL https://github.com/fltk/fltk/releases/download/release-1.4.5/fltk-1.4.5-source.tar.gz
      URL_HASH SHA256=eede1fb2b8e9c2e581e77082e15252145855c79aad30070ee3b24aabe2f926f1
      DOWNLOAD_EXTRACT_TIMESTAMP TRUE
    )
    FetchContent_MakeAvailable(fltk)
    set_property(DIRECTORY "${fltk_SOURCE_DIR}" PROPERTY EXCLUDE_FROM_ALL TRUE)
  endif()

  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" APPLICATION_LICENSE)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY.md" THIRD_PARTY_NOTICES)
  file(READ "${fltk_SOURCE_DIR}/COPYING" FLTK_LICENSE)
  configure_file(packaging/licenses.hpp.in generated/licenses.hpp @ONLY)
  add_executable(power-timer WIN32 src/main.cpp)
  target_include_directories(power-timer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(power-timer PRIVATE power_backend fltk::fltk Threads::Threads)
  power_timer_warnings(power-timer)
  if(POWER_TIMER_BUILD_TESTS)
    add_test(NAME cli_requires_dry_run COMMAND power-timer --smoke-test)
    add_test(NAME cli_invalid_language COMMAND power-timer --lang invalid)
    add_test(NAME cli_license COMMAND power-timer --license)
    set_tests_properties(cli_license PROPERTIES
      PASS_REGULAR_EXPRESSION "MIT License" TIMEOUT 5)
    set_tests_properties(cli_requires_dry_run cli_invalid_language PROPERTIES
      WILL_FAIL TRUE TIMEOUT 5)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
      find_program(XVFB_RUN xvfb-run)
    endif()
    foreach(language en ru)
      if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND XVFB_RUN)
        add_test(NAME gui_${language} COMMAND "${XVFB_RUN}" -a
          $<TARGET_FILE:power-timer> --dry-run --smoke-test --lang ${language})
      elseif(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        add_test(NAME gui_${language} COMMAND power-timer
          --dry-run --smoke-test --lang ${language})
      else()
        continue()
      endif()
      set_tests_properties(gui_${language} PROPERTIES TIMEOUT 20 LABELS gui)
    endforeach()
  endif()
  if(WIN32)
    enable_language(RC)
    configure_file(packaging/windows-version.rc.in windows-version.rc @ONLY)
    target_sources(power-timer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/windows-version.rc")
    if(MSVC)
      target_sources(power-timer PRIVATE packaging/windows.manifest)
    else()
      target_sources(power-timer PRIVATE packaging/windows.rc)
      target_include_directories(power-timer PRIVATE packaging)
    endif()
    set_target_properties(power-timer PROPERTIES OUTPUT_NAME PowerTimer)
  elseif(APPLE)
    enable_language(OBJCXX)
    target_sources(power-timer PRIVATE src/activity_macos.mm)
    set_source_files_properties(src/activity_macos.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
    target_link_libraries(power-timer PRIVATE "-framework Foundation")
    set_target_properties(power-timer PROPERTIES OUTPUT_NAME PowerTimer)
    # Keep the permission description and identity inside the standalone binary.
    configure_file(packaging/Info.plist.in PowerTimer-Info.plist @ONLY)
    target_link_options(power-timer PRIVATE
      "LINKER:-sectcreate,__TEXT,__info_plist,${CMAKE_CURRENT_BINARY_DIR}/PowerTimer-Info.plist")
    set_property(TARGET power-timer APPEND PROPERTY LINK_DEPENDS
      "${CMAKE_CURRENT_BINARY_DIR}/PowerTimer-Info.plist")
  endif()
  install(TARGETS power-timer RUNTIME DESTINATION . COMPONENT Runtime)
endif()
